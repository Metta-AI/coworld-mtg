1440: /// CR 603.4 + CR 603.6 + CR 603.10: Evaluate a trigger condition whose
1441: /// subject is the object from a zone-change event.
1442: ///
1443: /// Enter-the-battlefield conditions evaluate the live object in the destination
1444: /// zone. Death/leaves-the-battlefield conditions evaluate the zone-change
1445: /// record, which carries the event-time public characteristics used for LKI.
1446: pub fn matches_zone_change_event_object_filter(
1447:     state: &GameState,
1448:     event: &crate::types::events::GameEvent,
1449:     origin: Option<Zone>,
1450:     destination: Zone,
1451:     filter: &TargetFilter,
1452:     ctx: &FilterContext<'_>,
1453: ) -> bool {
1454:     let crate::types::events::GameEvent::ZoneChanged {
1455:         object_id,
1456:         from,
1457:         to,
1458:         record,
1459:     } = event
1460:     else {
1461:         return false;
1462:     };
1463: 
1464:     if origin.is_some_and(|required| *from != Some(required)) || *to != destination {
1465:         return false;
1466:     }
1467: 
1468:     if destination == Zone::Battlefield {
1469:         // CR 603.4: the intervening-if is rechecked when the ability resolves.
1470:         // CR 608.2h: a filter that reads the entrant's characteristics uses its
1471:         // CURRENT info only while the entrant is still in the public zone it was
1472:         // expected in (the battlefield); once it has left, it uses the entrant's
1473:         // LAST KNOWN INFORMATION. CR 603.10a: an ETB trigger is not a look-back
1474:         // trigger, so the normal CR 608.2h rule applies. CR 400.7: the live
1475:         // object is reverted to its base characteristics on its zone exit
1476:         // (zones.rs revert_layered_characteristics_to_base), so reading the live
1477:         // object after it leaves would compare against baseline P/T rather than
1478:         // its last on-battlefield values — hence the LKI dispatch below. A
1479:         // fully-absent entrant (objects.get == None) likewise routes to LKI
1480:         // rather than a spurious false from matches_target_filter.
1481:         // CR 400.7: a leave + re-entry reuses the SAME ObjectId but bumps the
1482:         // object's incarnation (move_to_zone -> reset_for_battlefield_entry). The
1483:         // original ETB trigger must use the ORIGINAL entrant's info, so the live
1484:         // object only counts as "still the original entrant" when its incarnation
1485:         // matches the one captured in the ETB record. A re-entered incarnation is
1486:         // a different object for this trigger and routes to the original exit LKI
1487:         // below. `entered_incarnation == None` (legacy/defensive records) falls
1488:         // back to the zone-only check.
1489:         let still_on_battlefield = state.objects.get(object_id).is_some_and(|obj| {
1490:             obj.zone == Zone::Battlefield
1491:                 && record
1492:                     .entered_incarnation
1493:                     .is_none_or(|inc| obj.incarnation == inc)
1494:         });
1495:         if still_on_battlefield {
1496:             matches_target_filter(state, *object_id, filter, ctx)
1497:         } else if let Some(lki) = state.lki_cache.get(object_id) {
1498:             // CR 608.2h: the entrant has left the battlefield — evaluate against
1499:             // its exit-time LKI (the most-recently-existed battlefield
1500:             // characteristics, snapshotted before the base revert).
1501:             matches_target_filter_on_lki_snapshot(state, *object_id, lki, filter, ctx)
1502:         } else {
1503:             // No exit LKI cached (defensive — a battlefield exit always caches
1504:             // one). Use the zone-change record rather than the reverted live
1505:             // object so the comparison never regresses to baseline P/T.
1506:             matches_target_filter_on_zone_change_record(state, record, filter, ctx)
1507:         }
1508:     } else {
1509:         matches_target_filter_on_zone_change_record(state, record, filter, ctx)
1510:     }
1511: }
